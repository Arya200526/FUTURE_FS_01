function searchMenu() {
    let input = document.getElementById("search").value.toLowerCase();
    let results = document.getElementById("searchResults");

    let items = ["Dashboard", "Profile", "Attendance", "Courses"];

    results.innerHTML = "";

    if (input === "") return;

    items.forEach(item => {
        if (item.toLowerCase().includes(input)) {
            let div = document.createElement("div");
            div.innerText = item;

            div.onclick = function() {
                showSection(item.toLowerCase());
                results.innerHTML = "";
                document.getElementById("search").value = "";
            };

            results.appendChild(div);
        }
    });
}

<ul>
    <li onclick="showSection('dashboard')">Dashboard</li>
    <li onclick="showSection('profile')">Profile</li>
    <li onclick="showSection('attendance')">Attendance</li>
    <li onclick="showSection('courses')">Courses</li>
</ul>
